# ---- Cell A2+A3 gop lam 1 : tai TUNG environment -> resize ngay -> xoa raw ----------------
# NGUYEN NHAN THAT SU cua 2 lan bi "No space left on device" truoc do: RAW tung duoc
# dat trong /kaggle/working, noi bi gioi han CUNG ~20GB (chung quota voi chinh output

# ~60GB, KHONG tinh vao gioi han 20GB do. /kaggle/working tu gio chi chua ban da resize
# (~13GB, an toan). Van giu them tai tung difficulty rieng + delete_zip=True lam lop
# an toan phu (giam dinh dung luong can trong /kaggle/temp, du no da rong rai hon nhieu).
import glob, pathlib, shutil
from PIL import Image
from concurrent.futures import ThreadPoolExecutor
import tartanair as ta

RAW = "/kaggle/temp/_raw"       # KHONG dat trong /kaggle/working - cho do bi gioi han cung ~20GB
                                 # (chung quota voi output cuoi cung). /kaggle/temp la o dia
                                 # scratch RIENG cua Kaggle, ~60GB, khong tinh vao gioi han do.
DST = pathlib.Path("/kaggle/working/tartanair-v2")   # <- data_root cho train
ta.init(tartanair_root=RAW)

ENVS = [
    # 7 environment goc
    "AmericanDiner", "ArchVizTinyHouseDay", "CountryHouse", "DesertGasStation",
    "OldTownNight", "RetroOffice", "WaterMillDay",
    # 7 environment moi them de tang da dang canh (~26GB raw)
    "HongKong", "OldTownSummer", "House", "CoalMine", "Office",
    "HQWesternSaloon", "VictorianStreet",
]

def conv(p, src):
    p = pathlib.Path(p)
    if not p.is_file():
        return
    suffix = p.suffix.lower()
    if suffix not in (".png", ".txt"):
        return   # bo qua .zip va bat ky file la nao khac - KHONG dung "else: copy tat ca"
                 # (bug cu: neu zip chua kip xoa luc quet, no bi copy nham vao dataset)
    rel = p.relative_to(src)
    out = DST / rel
    out.parent.mkdir(parents=True, exist_ok=True)
    if suffix == ".png":
        with Image.open(p) as im:
            im.convert("RGB").resize((256, 256), Image.BILINEAR).save(out)
    else:  # acc.txt / gyro.txt / imu_time.txt / cam_time.txt / pose_lcam_front.txt
        shutil.copy2(p, out)

for env in ENVS:
    # Tai TUNG difficulty rieng (khong easy+hard cung luc) de dinh dinh dung luong raw
    # dinh trong luc tai chi con ~1 nua 1 env - da tung bi "No space left on device"
    # khi tai WaterMillDay (10GB) cung luc ca easy+hard.
    for diff in ["easy", "hard"]:
        ta.download(
            env=[env],
            difficulty=[diff],
            modality=["image", "imu"],
            camera_name=["lcam_front"],
            unzip=True,
            delete_zip=True,   # xoa tung zip ngay sau khi giai nen no, khong doi het rieng loat xong
        )
        src = pathlib.Path(RAW)
        files = glob.glob(str(src / "**" / "*"), recursive=True)
        with ThreadPoolExecutor(8) as ex:
            list(ex.map(lambda p: conv(p, src), files))
        shutil.rmtree(RAW, ignore_errors=True)
    print("done", env, flush=True)
    !df -h /kaggle/working /kaggle/temp   # theo doi dinh trong con lai sau moi env

================================================================================
 D. GHI CHU
================================================================================
- Quota GPU Kaggle: ~30h/tuan, moi phien toi da 12h. Notebook A khong dung GPU.
- /kaggle/working gioi han output ~20GB -> vi vay Cell A3 resize + xoa raw.
- num_workers: notebook GPU co 4 vCPU -> 4 la du.
- batch_size goc trong base.yaml la 64 (cho GPU >= 24GB). Tren Kaggle dung 32 qua CLI.
- Neu muon batch hieu dung = 64 ma van 16GB: can them gradient accumulation vao
  qwt_jepa/train/engine.py (chua co) - noi neu can lam.
- Kiem tra dong "[eval eN] L_jepa .. PSNR .. RMSE .." sau moi epoch de theo doi that su;
  so trong vong lap train la theo batch nen nhieu manh.
- Canh bao collapse: theo doi z_std; neu tut dan ve ~0.0x thi tang train.ema.start
  (0.998-0.999) trong kaggle.yaml. Luc khoi tao z_std ~0.246 (do tren RTX 4060).
- Khoi `corruption:` da doi han (profile DRONE, low-light, khong quy luat). Xem chu thich
  trong base.yaml. Cac nut chinh:
    couple / condition : rang buoc do sang <-> do nhoe <-> muc nhieu; `condition` san > 0
                         de khong sinh ra cap (sach, gan nhu sach)
    exposure           : da so anh TOI di ~10% -> model hoc TANG DO SANG
    alias_freq/p_alias : rung rotor drone (70-280 Hz) gap tan so xuong bang 100 Hz
    burst/step/stuck   : nhieu IMU khong deu theo thoi gian
- LUU Y ve metric: exposure lam anh toi, nen PSNR trong log gop CA bu sang lan khu nhieu.
  Khong so PSNR voi cac lan chay truoc ngay 2026-09-09.
- Cat ngan train/eval + early stop (base.yaml, muc train:):
    limit_train_batches: 500   # so batch/epoch (0 = full). Epoch time ~ ti le voi so nay
    limit_val_batches:   100   # so batch eval/epoch (0 = full)
    early_stop: {metric: L_jepa, mode: min, patience: 6, min_delta: 0.001}
      -> dung khi val metric khong cai thien qua min_delta trong `patience` epoch lien tiep.
      -> metric doi thanh psnr thi phai dat mode: max.
      -> patience: 0 de TAT early stop.
    best.pt luu theo dung metric early_stop (khong con co dinh L_jepa).
