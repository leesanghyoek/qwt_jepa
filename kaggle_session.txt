import glob, pathlib, shutil, io, contextlib
from PIL import Image
from concurrent.futures import ThreadPoolExecutor
import tartanair as ta

RAW = "/kaggle/temp/_raw"       # KHONG dat trong /kaggle/working - cho do bi gioi han cung ~20GB
                                 # (chung quota voi output cuoi cung). /kaggle/temp la o dia
                                 # scratch RIENG cua Kaggle, ~60GB, khong tinh vao gioi han do.
DST = pathlib.Path("/kaggle/working/tartanair-v2")   # <- data_root cho train
ta.init(tartanair_root=RAW)

ENVS = [
    # 7 environment goc
    "AmericanDiner", "ArchVizTinyHouseDay", "CountryHouse", "DesertGasStation",
    "OldTownNight", "RetroOffice", "WaterMillDay",
    # 7 environment moi them de tang da dang canh (~26GB raw)
    "HongKong", "OldTownSummer", "House", "CoalMine", "Office",
    "HQWesternSaloon", "VictorianStreet",
]

def conv(p, src):
    p = pathlib.Path(p)
    if not p.is_file():
        return
    suffix = p.suffix.lower()
    if suffix not in (".png", ".txt"):
        return   # bo qua .zip va bat ky file la nao khac - KHONG dung "else: copy tat ca"
                 # (bug cu: neu zip chua kip xoa luc quet, no bi copy nham vao dataset)
    rel = p.relative_to(src)
    out = DST / rel
    out.parent.mkdir(parents=True, exist_ok=True)
    if suffix == ".png":
        with Image.open(p) as im:
            im.convert("RGB").resize((256, 256), Image.BILINEAR).save(out)
    else:  # acc.txt / gyro.txt / imu_time.txt / cam_time.txt / pose_lcam_front.txt
        shutil.copy2(p, out)

for env in ENVS:
    # Tai TUNG difficulty rieng (khong easy+hard cung luc) de dinh dinh dung luong raw
    # dinh trong luc tai chi con ~1 nua 1 env - da tung bi "No space left on device"
    # khi tai WaterMillDay (10GB) cung luc ca easy+hard.
    for diff in ["easy", "hard"]:
        # ta.download() in tien do TUNG FILE -> hang nghin dong, treo trinh duyet.
        # Nuot output khi THANH CONG; neu loi thi moi in 3000 ky tu cuoi de con debug.
        _log = io.StringIO()
        try:
            with contextlib.redirect_stdout(_log):
                ta.download(
                    env=[env],
                    difficulty=[diff],
                    modality=["image", "imu"],
                    camera_name=["lcam_front"],
                    unzip=True,
                    delete_zip=True,   # xoa tung zip ngay sau khi giai nen, khong doi het loat
                )
        except Exception:
            print(_log.getvalue()[-3000:])
            raise
        src = pathlib.Path(RAW)
        files = glob.glob(str(src / "**" / "*"), recursive=True)
        with ThreadPoolExecutor(8) as ex:
            list(ex.map(lambda p: conv(p, src), files))
        shutil.rmtree(RAW, ignore_errors=True)
    free = {d: shutil.disk_usage(d)[2] / 2**30 for d in ("/kaggle/working", "/kaggle/temp")}
    print(f"done {env:22} | working con {free['/kaggle/working']:5.1f} GB"
          f" | temp con {free['/kaggle/temp']:5.1f} GB", flush=True)   # 1 dong thay vi df -h 6 dong

print("resize xong ->", DST)

# val PSNR, mode max - KHONG con la L_jepa). Uoc tinh tren Kaggle (4 vCPU nen dataloader la nut co chai, ~130 sample/s):
#   500 batch x 32 = 16000 sample/epoch -> ~2 phut train + ~25s eval = ~2.5 phut/epoch

# nay VE DE LEN CHO CU (clear_output) nen so dong tren man hinh KHONG BAO GIO TANG,

import subprocess, collections, time
from IPython.display import clear_output

CMD = ["torchrun", "--standalone", "--nproc_per_node=2", "qwt_jepa/train/train_ddp.py",
       "--config", "qwt_jepa/configs/kaggle.yaml",
       "--out", "/kaggle/working/runs/base",
       "--batch-size", "16", "--num-workers", "2"]   # batch MOI GPU -> hieu dung 32
TAIL      = 20                          # so dong log gan nhat giu tren man hinh
REFRESH_S = 0.7                         # toi thieu bao lau moi ve lai (tranh spam DOM)
LOGFILE   = "/kaggle/working/train.log"

KEY = ("[eval", "resume tu", "train ", "train:", "norm stats", "QwtJepa",
       "best.pt", "early-stop", "dung som", "xong.", "canh bao", "Error", "Traceback")
tail, marks = collections.deque(maxlen=TAIL), []
t0 = last = time.time()

def draw(done=False):
    clear_output(wait=True)
    if marks:
        print("=== MOC QUAN TRONG (giu lai het) ===")
        print("\n".join(marks[-14:]))
        print()
    print(f"=== {TAIL} DONG GAN NHAT   |   da chay {(time.time()-t0)/60:.1f} phut ===")
    print("\n".join(tail))
    print(f"\nlog day du: {LOGFILE}" + ("   [DA XONG]" if done else ""))

# QUAN TRONG: phai boc try/finally va GIET tien trinh con.
# Neu khong: bam Stop o notebook chi nem KeyboardInterrupt vao Python cua notebook,
# con `proc` (train.py) VAN CHAY NEN va giu VRAM. Chay lai cell luc do sinh ra tien
# trinh THU HAI, hai ban tranh GPU, cham nhu ru va co the OOM. Dau hieu: log in ra
# hai lan "QwtJepa: ... params train", hoac nvidia-smi thay hai tien trinh python.
proc = None
try:
    with open(LOGFILE, "w") as lf:
        proc = subprocess.Popen(CMD, stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                                text=True, bufsize=1)
        for line in proc.stdout:
            line = line.rstrip()
            lf.write(line + "\n"); lf.flush()  # flush de xem duoc log ngay ca khi notebook chet
            tail.append(line)
            if any(k in line for k in KEY):
                marks.append(line)
            if time.time() - last > REFRESH_S:
                draw(); last = time.time()
        proc.wait()
finally:
    if proc is not None and proc.poll() is None:
        proc.terminate()
        try:
            proc.wait(timeout=10)
        except subprocess.TimeoutExpired:
            proc.kill(); proc.wait()
        print("\n[da dung tien trinh train]")
draw(done=True)
print("ma thoat:", proc.returncode if proc else "?")

================================================================================
 C2. THEO DOI z_std - VIEC QUAN TRONG NHAT KHI TRAIN
================================================================================
Model nay tung collapse toan bo trong epoch 0. Dau hieu nam o z_std, khong nam o loss.

Sau moi epoch, dong [eval eN] in ra z_std (chinh la std cua z_ctx tren tap val):

  LANH:  z_std leo deu va khong tut. z_std day la std THEO NOI DUNG (doi anh thi bieu
         dien doi bao nhieu), khong phai std gop ca (batch, token).
  HONG:  z_std tut dan ve 0.1  -> collapse, dung train, bao lai

Dong eval con in "L_jepa 0.0510/0.0569pos". So thu hai la L_jepa ma mot ke gian lan
CHI doan theo vi tri token (khong nhin anh) dat duoc. DOC THEO TI SO:

  ti so < 1.0  -> JEPA that su dung thong tin anh          OK
  ti so >= 1.0 -> JEPA khong hoc gi ve noi dung, dung lai   HONG

Ti so nay thuong bat dau rat cao (~15-40) roi tut xuong duoi 1.0 trong vai epoch dau.
Day la chi so KHO LACH nhat: no hoi thang "bo anh di thi co te hon khong?".

Ba dieu de KHONG hoang mang:
  1. L_jepa co the TANG trong vai epoch dau. Do la dau hieu TOT: z_ctx dang bung ra,
     z_tgt bam theo qua EMA, bai toan du doan tro nen kho THAT SU. Chi lo khi z_std tut.
  2. L_var (trong dong log train) tut dan ve ~0 nghia la rang buoc chong collapse da
     duoc thoa man. Neu no ket tren 0.3 nhieu epoch thi tang train.loss.lambda_var.
  3. PSNR ~3-4 dB luc dau la binh thuong (recon head chua hoc gi, dau ra khong bi chan).

Cac moc PSNR de doi chieu (do that tren tap valid, xem LOG_TRAIN_GIAI_THICH.md):
  14.0 dB  = doan mau trung binh cua anh   (san vo dung)
  20.9 dB  = chinh anh nhieu dau vao       (nguong PHAI vuot)
  24.5 dB  = tran kien truc voi masking hien tai

HAI CHO DE SAI:

1. --batch-size la batch MOI GPU. Dat 16 x 2 GPU = batch hieu dung 32, DUNG BANG cac
   run 1 GPU truoc day -> so sanh truc tiep duoc, khong phai chinh lr.
   Neu dat 32/GPU thi batch hieu dung la 64 -> phai nhan doi lr (1.5e-4 -> 3e-4).

2. --num-workers la MOI tien trinh. Kaggle co 4 vCPU, 2 tien trinh x 2 worker = 4.
   Dat 4/tien trinh la 8 worker tren 4 vCPU -> tranh CPU, CHAM HON.

KY VONG THUC TE - dung mong gap doi:
Uoc tu so lieu da do (1 GPU: 60 im/s truoc khi them luot thu 2, 48 im/s sau do):
    CPU nap du lieu toi da  ~60 im/s
    GPU (1 the)             ~48 im/s
    GPU (2 the)             ~96 im/s  -> nhung bi CPU chan o 60
=> thuc te ~48 -> ~60 im/s, tang ~25%. Nut co chai chuyen tu GPU sang DATALOADER.
Muon nhanh hon nua phai lam dataloader nhanh hon (anh nho hon, hoac cache san), chu
khong phai them GPU.

Dong log dau tien cung phai in: "world 2 | batch/gpu 16 | batch hieu dung 32"

    limit_train_batches: 500   # so batch/epoch (0 = full). Epoch time ~ ti le voi so nay

      -> dung khi val metric khong cai thien qua min_delta trong `patience` epoch lien tiep.
      -> metric doi thanh psnr thi phai dat mode: max.
      -> patience: 0 de TAT early stop.
    best.pt luu theo dung metric early_stop (khong con co dinh L_jepa).

--batch-size LA BATCH MOI GPU. Batch hieu dung = batch_size * so_gpu.
    1 GPU: --batch-size 32              -> hieu dung 32
    2 GPU: --batch-size 16              -> hieu dung 32  (giu nguyen lr)
    2 GPU: --batch-size 32              -> hieu dung 64  (phai x2 lr theo ti le tuyen tinh)
Playbook nay dung 16 de batch hieu dung khong doi -> khong phai chinh lr, va con so
so sanh truc tiep duoc voi cac lan chay 1 GPU truoc do.

limit_train_batches dem theo MOI rank: 500 batch x 16 x 2 GPU = 16000 sample/epoch,
bang het voi 1 GPU (500 x 32). total_steps khong doi -> lich LR/EMA khong doi.
limit_val_batches thi train_ddp.py TU CHIA cho so GPU (lim_val_rank), nen tong so mau
eval bang voi 1 GPU va PSNR so sanh truc tiep duoc.

num-workers 2 chu khong phai 4: notebook chi co 4 vCPU va gio co 2 tien trinh.

KHONG resume checkpoint 1 GPU bang lenh 2 GPU (va nguoc lai): batch hieu dung doi ->
lich LR/EMA lech. Doi so GPU thi train lai tu dau.
